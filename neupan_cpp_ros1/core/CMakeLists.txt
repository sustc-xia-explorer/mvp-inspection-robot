cmake_minimum_required(VERSION 3.16)
project(neupan_ros1_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(NEUPAN_ROS1_CORE_TESTS "Build core regression and omni parity tests" ON)
find_package(Eigen3 3.3 REQUIRED)
find_package(OsqpEigen REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(neupan_ros1_core STATIC
  src/mlp.cpp src/robot.cpp src/dune.cpp src/nrmp.cpp src/initial_path.cpp
  src/pan.cpp src/neupan_planner.cpp src/tensor_io.cpp)
target_include_directories(neupan_ros1_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(neupan_ros1_core PUBLIC Eigen3::Eigen OsqpEigen::OsqpEigen yaml-cpp)
target_compile_options(neupan_ros1_core PRIVATE -Wall -Wextra)
add_executable(neupan_ros1_sim_backend tools/sim_backend.cpp)
target_link_libraries(neupan_ros1_sim_backend PRIVATE neupan_ros1_core)
if(NEUPAN_ROS1_CORE_TESTS)
  enable_testing()
  find_package(GTest REQUIRED)
  set(_tests test_dune test_nrmp test_replay test_command_limits)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/../test/core/test_omni.cpp)
    list(APPEND _tests test_omni)
  endif()
  foreach(t IN LISTS _tests)
    add_executable(ros1_${t} ../test/core/${t}.cpp)
    target_link_libraries(ros1_${t} PRIVATE neupan_ros1_core GTest::gtest GTest::gtest_main)
    target_compile_definitions(ros1_${t} PRIVATE
      NEUPAN_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../test/data"
      NEUPAN_MODEL_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../models"
      NEUPAN_CONFIG_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../config")
    add_test(NAME ros1_${t} COMMAND ros1_${t})
  endforeach()
endif()
